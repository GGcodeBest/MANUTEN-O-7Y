const SUPABASE_ANON_KEY = 'SUA-CHAVE-ANONIMA-DO-SUPABASE';

/* --------------------------------------------------------------------------
   1. NAVEGAÇÃO E INTERFACE
   -------------------------------------------------------------------------- */
function initNavigation() {
    // Menu Lateral - Troca de Abas
    const navItems = document.querySelectorAll('.sidebar-nav li');
    navItems.forEach(item => {
        item.addEventListener('click', () => {
            const tabId = item.getAttribute('data-tab');
            switchTab(tabId);
        });
    });

    // Toggle Sidebar Collapsed
    document.getElementById('toggleSidebar').addEventListener('click', () => {
        document.getElementById('sidebar').classList.toggle('collapsed');
    });

    // Busca Rápida
    document.getElementById('quickSearch').addEventListener('keypress', (e) => {
        if (e.key === 'Enter') {
            const term = e.target.value.trim().toLowerCase();
            if (term) {
                switchTab('pesquisa');
                document.getElementById('fPlaca').value = term;
                executarPesquisaAvançada();
            }
        }
    });
}

function switchTab(tabId) {
    document.querySelectorAll('.sidebar-nav li').forEach(li => li.classList.remove('active'));
    document.querySelectorAll('.tab-content').forEach(tc => tc.classList.remove('active'));

    const navLi = document.querySelector(`.sidebar-nav li[data-tab="${tabId}"]`);
    if (navLi) navLi.classList.add('active');

    const targetTab = document.getElementById(`tab-${tabId}`);
    if (targetTab) targetTab.classList.add('active');

    if (tabId === 'dashboard') atualizarDashboard();
}

function openModal(modalId) {
    preencherSelectsFormularios();
    document.getElementById(modalId).style.display = 'flex';
}

function closeModal(modalId) {
    document.getElementById(modalId).style.display = 'none';
    if(modalId === 'modalOcorrencia') {
        document.getElementById('alertaRecorrencia').classList.add('hidden');
    }
}

/* --------------------------------------------------------------------------
   2. CARREGAMENTO E SINCRONIZAÇÃO DE DADOS (SUPABASE / LOCAL)
   -------------------------------------------------------------------------- */
async function carregarDadosIniciais() {
    if (supabase) {
        try {
            const { data: veiculos } = await supabase.from('veiculos').select('*');
            const { data: ocorrencias } = await supabase.from('ocorrencias').select('*');
            const { data: ordens } = await supabase.from('ordens_servico').select('*');
            const { data: mecanicos } = await supabase.from('mecanicos').select('*');

            db.veiculos = veiculos || [];
            db.ocorrencias = ocorrencias || [];
            db.ordens = ordens || [];
            db.mecanicos = mecanicos || [];
        } catch (error) {
            console.error("Erro ao carregar do Supabase, utilizando modo local:", error);
            carregarDadosDemonstracao();
        }
    } else {
        carregarDadosDemonstracao();
    }

    renderizarTabelas();
    atualizarDashboard();
}

/* --------------------------------------------------------------------------
   3. RENDERING DAS TABELAS
   -------------------------------------------------------------------------- */
function renderizarTabelas() {
    renderTabelaVeiculos();
    renderTabelaOcorrencias();
    renderTabelaOrdens();
    renderTabelaMecanicos();
}

function renderTabelaVeiculos() {
    const tbody = document.getElementById('tblVeiculos');
    tbody.innerHTML = db.veiculos.map(v => `
        <tr>
            <td><strong>${v.placa}</strong></td>
            <td>${v.prefixo}</td>
            <td>${v.marca} / ${v.modelo}</td>
            <td>${v.ano}</td>
            <td>${v.km_atual.toLocaleString()} KM</td>
            <td>${v.unidade}</td>
            <td>${v.motorista_habitual || '-'}</td>
            <td><span class="badge ${getBadgeSituacao(v.situacao)}">${v.situacao.replace('_', ' ')}</span></td>
            <td>
                <button class="btn btn-sm btn-secondary" onclick="verHistoricoVeiculo('${v.id}')"><i class="fa-solid fa-clock-rotate-left"></i> Histórico</button>
            </td>
        </tr>
    `).join('');
}

function renderTabelaOcorrencias() {
    const tbody = document.getElementById('tblOcorrencias');
    tbody.innerHTML = db.ocorrencias.map(o => {
        const v = db.veiculos.find(x => x.id === o.veiculo_id);
        return `
            <tr>
                <td>#${o.numero_ocorrencia}</td>
                <td>${o.data_hora.substring(0,10)}</td>
                <td>${v ? v.placa + ' (' + v.prefixo + ')' : '-'}</td>
                <td>${o.motorista}</td>
                <td><span class="badge badge-info">${o.categoria.toUpperCase()}</span></td>
                <td><span class="badge ${o.prioridade === 'critica' ? 'badge-danger' : 'badge-warning'}">${o.prioridade}</span></td>
                <td><span class="badge badge-secondary">${o.status}</span></td>
                <td>
                    <button class="btn btn-sm btn-primary" onclick="iniciarOSApartirOcorrencia('${o.id}')"><i class="fa-solid fa-wrench"></i> Gerar OS</button>
                </td>
            </tr>
        `;
    }).join('');
}

function renderTabelaOrdens() {
    const tbody = document.getElementById('tblOrdens');
    tbody.innerHTML = db.ordens.map(os => {
        const v = db.veiculos.find(x => x.id === os.veiculo_id);
        const m = db.mecanicos.find(x => x.id === os.mecanico_id);
        return `
            <tr>
                <td><strong>OS #${os.numero_os}</strong></td>
                <td>${os.ocorrencia_id ? '#' + os.ocorrencia_id : 'Direta'}</td>
                <td>${v ? v.placa : '-'}</td>
                <td>${os.data_entrada}</td>
                <td>${m ? m.nome : os.oficina_terceira || '-'}</td>
                <td>R$ ${parseFloat(os.valor_total || 0).toFixed(2)}</td>
                <td>${os.garantia_dias ? os.garantia_dias + ' dias' : 'Sem garantia'}</td>
                <td><span class="badge badge-success">${os.status}</span></td>
                <td>
                    <button class="btn btn-sm btn-secondary" onclick="imprimirOS('${os.id}')"><i class="fa-solid fa-print"></i> Imprimir</button>
                </td>
            </tr>
        `;
    }).join('');
}

function renderTabelaMecanicos() {

function getBadgeSituacao(situacao) {
    if (situacao === 'disponivel') return 'badge-success';
    if (situacao === 'em_manutencao') return 'badge-warning';
    return 'badge-danger';
}

/* --------------------------------------------------------------------------
   4. MECANISMO DE IDENTIFICAÇÃO DE RECORRÊNCIAS (ALERTA DE PROBLEMAS)
   -------------------------------------------------------------------------- */
function verificarRecorrenciaEmTempoReal() {
    const veiculoId = document.getElementById('oVeiculoId').value;
    const categoria = document.getElementById('oCategoria').value;
    const alertaBox = document.getElementById('alertaRecorrencia');
    const alertaConteudo = document.getElementById('alertaRecorrenciaConteudo');

    if (!veiculoId || !categoria) {
        alertaBox.classList.add('hidden');
        return;
    }

    // Buscar ordens de serviço anteriores do mesmo veículo na mesma categoria
    const ocorrenciasAnteriores = db.ocorrencias.filter(o => o.veiculo_id === veiculoId && o.categoria === categoria);
    
    if (ocorrenciasAnteriores.length > 0) {
        const ultimaOcorrencia = ocorrenciasAnteriores[ocorrenciasAnteriores.length - 1];
        const ultimaOS = db.ordens.find(os => os.ocorrencia_id === ultimaOcorrencia.id);
        const mec = ultimaOS ? db.mecanicos.find(m => m.id === ultimaOS.mecanico_id) : null;

        alertaBox.classList.remove('hidden');
        alertaConteudo.innerHTML = `
            <p>Este caminhão já apresentou um problema na categoria <strong>${categoria.toUpperCase()}</strong> anteriormente.</p>
            <ul>
                <li><strong>Data do último registro:</strong> ${ultimaOcorrencia.data_hora.substring(0, 10)}</li>
                <li><strong>Diagnóstico anterior:</strong> ${ultimaOS ? ultimaOS.diagnostico_tecnico || 'N/A' : 'Em aberto'}</li>
                <li><strong>Mecânico responsável:</strong> ${mec ? mec.nome : 'N/A'}</li>
                <li><strong>Valor pago anteriormente:</strong> R$ ${ultimaOS ? parseFloat(ultimaOS.valor_total).toFixed(2) : '0.00'}</li>
                <li><strong>Ocorrências registradas nesta categoria:</strong> ${ocorrenciasAnteriores.length} vez(es)</li>
            </ul>
        `;
    } else {
        alertaBox.classList.add('hidden');
    }
}

/* --------------------------------------------------------------------------
   5. CADASTROS E SALVAMENTO DE DADOS
   -------------------------------------------------------------------------- */
async function salvarVeiculo(e) {
    e.preventDefault();
    const novoVeiculo = {
        placa: document.getElementById('vPlaca').value.toUpperCase(),
        prefixo: document.getElementById('vPrefixo').value,
        marca: document.getElementById('vMarca').value,
        modelo: document.getElementById('vModelo').value,
        ano: parseInt(document.getElementById('vAno').value),
        km_atual: parseInt(document.getElementById('vKm').value),
        tipo: document.getElementById('vTipo').value,
        unidade: document.getElementById('vUnidade').value,
        motorista_habitual: document.getElementById('vMotorista').value,
        situacao: document.getElementById('vSituacao').value,
        observacoes: document.getElementById('vObservacoes').value
    };

    if (supabase) {
        const { data, error } = await supabase.from('veiculos').insert([novoVeiculo]).select();
        if (data) db.veiculos.push(data[0]);
    } else {
        novoVeiculo.id = Date.now().toString();
        db.veiculos.push(novoVeiculo);
    }

    closeModal('modalVeiculo');
    renderTabelaVeiculos();
    atualizarDashboard();
    alert('Veículo cadastrado com sucesso!');
}

async function salvarOcorrencia(e) {
    e.preventDefault();
    const novaOcorrencia = {
        veiculo_id: document.getElementById('oVeiculoId').value,
        motorista: document.getElementById('oMotorista').value,
        km_momento: parseInt(document.getElementById('oKm').value),
        categoria: document.getElementById('oCategoria').value,
        prioridade: document.getElementById('oPrioridade').value,
        localizacao: document.getElementById('oLocal').value,
        descricao: document.getElementById('oDescricao').value,
        sintomas: document.getElementById('oSintomas').value,
        status: 'aberta',
        data_hora: new Date().toISOString()
    };

    if (supabase) {
        const { data } = await supabase.from('ocorrencias').insert([novaOcorrencia]).select();
        if (data) db.ocorrencias.push(data[0]);
    } else {
        novaOcorrencia.id = Date.now().toString();
        novaOcorrencia.numero_ocorrencia = db.ocorrencias.length + 101;
        db.ocorrencias.push(novaOcorrencia);
    }

    // Atualiza status do veículo para em_manutencao
    const veiculo = db.veiculos.find(v => v.id === novaOcorrencia.veiculo_id);
    if (veiculo) veiculo.situacao = 'em_manutencao';

    closeModal('modalOcorrencia');
    renderTabelaOcorrencias();
    renderTabelaVeiculos();
    atualizarDashboard();
    alert('Ocorrência registrada com sucesso!');
}

function calcularTotalOS() {
    const mo = parseFloat(document.getElementById('osValorMaoObra').value) || 0;
    const pe = parseFloat(document.getElementById('osValorPecas').value) || 0;
    const ou = parseFloat(document.getElementById('osOutrosCustos').value) || 0;
    document.getElementById('osValorTotal').value = (mo + pe + ou).toFixed(2);
}

async function salvarOS(e) {
    e.preventDefault();
    const novaOS = {
        ocorrencia_id: document.getElementById('osOcorrenciaId').value || null,
        veiculo_id: document.getElementById('osVeiculoId').value,
        mecanico_id: document.getElementById('osMecanicoId').value,
        oficina_terceira: document.getElementById('osOficinaTerceira').value,
        data_entrada: document.getElementById('osDataEntrada').value,
        data_conclusao: document.getElementById('osDataConclusao').value,
        km_reparo: parseInt(document.getElementById('osKmReparo').value),
        responsavel_aprovacao: document.getElementById('osResponsavelAprovacao').value,
        diagnostico_tecnico: document.getElementById('osDiagnostico').value,
        descricao_reparo: document.getElementById('osDescricaoReparo').value,
        valor_maos_obra: parseFloat(document.getElementById('osValorMaoObra').value) || 0,
        valor_pecas: parseFloat(document.getElementById('osValorPecas').value) || 0,
        outros_custos: parseFloat(document.getElementById('osOutrosCustos').value) || 0,
        valor_total: parseFloat(document.getElementById('osValorTotal').value) || 0,
        garantia_dias: parseInt(document.getElementById('osGarantiaDias').value) || 0,
        garantia_km: parseInt(document.getElementById('osGarantiaKm').value) || 0,
        teste_pos_reparo: document.getElementById('osTesteRealizado').checked,
        resultado_teste: document.getElementById('osResultadoTeste').value,
        pendencias_pos_reparo: document.getElementById('osPendencias').value,
        status: 'concluido'
    };

    if (supabase) {
        const { data } = await supabase.from('ordens_servico').insert([novaOS]).select();
        if (data) db.ordens.push(data[0]);
    } else {
        novaOS.id = Date.now().toString();
        novaOS.numero_os = db.ordens.length + 501;
        db.ordens.push(novaOS);
    }

    // Se vinculado a ocorrencia, fecha ocorrencia
    if (novaOS.ocorrencia_id) {
        const oc = db.ocorrencias.find(o => o.id === novaOS.ocorrencia_id);
        if (oc) oc.status = 'concluida';
    }

    // Libera o veículo
    const veiculo = db.veiculos.find(v => v.id === novaOS.veiculo_id);
    if (veiculo) {
        veiculo.situacao = 'disponivel';
        veiculo.km_atual = novaOS.km_reparo;
    }

    closeModal('modalOS');
    renderTabelaOrdens();
    renderTabelaVeiculos();
    atualizarDashboard();
    alert('Ordem de Serviço finalizada com sucesso!');
}

/* --------------------------------------------------------------------------
   6. HISTÓRICO COMPLETO DO VEÍCULO (TIMELINE)
   -------------------------------------------------------------------------- */
function verHistoricoVeiculo(veiculoId) {
    const v = db.veiculos.find(x => x.id === veiculoId);
    if (!v) return;

    switchTab('historico-veiculo');

    document.getElementById('veiculoHeaderInfo').innerHTML = `
        <h2>${v.marca} ${v.modelo} - Placa: ${v.placa} (Prefixo: ${v.prefixo})</h2>
        <p><strong>Ano:</strong> ${v.ano} | <strong>KM Atual:</strong> ${v.km_atual.toLocaleString()} KM | <strong>Filial:</strong> ${v.unidade} | <strong>Motorista:</strong> ${v.motorista_habitual || 'Não definido'}</p>
    `;

    const ocorrenciasVeiculo = db.ocorrencias.filter(o => o.veiculo_id === veiculoId);
    const ordensVeiculo = db.ordens.filter(os => os.veiculo_id === veiculoId);

    const timeline = document.getElementById('timelineVeiculo');
    
    if (ocorrenciasVeiculo.length === 0 && ordensVeiculo.length === 0) {
        timeline.innerHTML = '<p>Nenhum histórico de manutenção registrado para este caminhão.</p>';
        return;
    }

    let html = '';
    ocorrenciasVeiculo.forEach(o => {
        const osRelacionada = ordensVeiculo.find(os => os.ocorrencia_id === o.id);
        html += `
            <div class="timeline-item">
                <h4><i class="fa-solid fa-triangle-exclamation text-orange"></i> Ocorrência #${o.numero_ocorrencia} - ${o.categoria.toUpperCase()} (${o.data_hora.substring(0,10)})</h4>
                <p><strong>Motorista:</strong> ${o.motorista} | <strong>KM:</strong> ${o.km_momento}</p>
                <p><strong>Descrição:</strong> ${o.descricao}</p>
                ${osRelacionada ? `
                    <div style="background:#f1f5f9; padding:10px; border-radius:6px; margin-top:8px;">
                        <strong>Conserto Relacionado (OS #${osRelacionada.numero_os}):</strong>
                        <p><strong>Diagnóstico:</strong> ${osRelacionada.diagnostico_tecnico}</p>
                        <p><strong>Reparo:</strong> ${osRelacionada.descricao_reparo}</p>
                        <p><strong>Valor Total:</strong> R$ ${parseFloat(osRelacionada.valor_total).toFixed(2)} \vert{} <strong>Garantia:</strong>${osRelacionada.garantia_dias} dias</p>
                    </div>
                ` : '<p><em>Aguardando início do reparo.</em></p>'}
            </div>
        `;
    });

    timeline.innerHTML = html;
}

/* --------------------------------------------------------------------------
   7. DASHBOARD E GRÁFICOS
   -------------------------------------------------------------------------- */
function atualizarDashboard() {
    document.getElementById('kpiTotalVeiculos').innerText = db.veiculos.length;
    document.getElementById('kpiEmManutencao').innerText = db.veiculos.filter(v => v.situacao === 'em_manutencao').length;
    document.getElementById('kpiAguardando').innerText = db.ocorrencias.filter(o => o.status === 'aberta').length;
    document.getElementById('kpiConcluidasMes').innerText = db.ordens.filter(os => os.status === 'concluido').length;
    document.getElementById('kpiOcorrenciasAbertas').innerText = db.ocorrencias.filter(o => o.status !== 'concluida').length;
    document.getElementById('kpiLiberados').innerText = db.veiculos.filter(v => v.situacao === 'disponivel').length;

    const gastoTotal = db.ordens.reduce((acc, os) => acc + parseFloat(os.valor_total || 0), 0);
    document.getElementById('kpiGastoMes').innerText = `R$ ${gastoTotal.toFixed(2)}`;

    atualizarGraficos();
}

function initChartJS() {
    const ctxGastos = document.getElementById('chartGastos').getContext('2d');
    chartGastos = new Chart(ctxGastos, {
        type: 'bar',
        data: { labels: ['Jul', 'Ago', 'Set', 'Out'], datasets: [{ label: 'Gastos (R$)', data: [12000, 15000, 8000, 18500], backgroundColor: '#0f172a' }] }
    });

    const ctxCat = document.getElementById('chartCategorias').getContext('2d');
    chartCategorias = new Chart(ctxCat, {
        type: 'pie',
        data: { labels: ['Motor', 'Freios', 'Elétrico', 'Câmbio'], datasets: [{ data: [40, 25, 20, 15], backgroundColor: ['#ef4444', '#f97316', '#3b82f6', '#10b981'] }] }
    });
}

function atualizarGraficos() {
    // Atualização dinâmica dos dados nos gráficos
    if (chartGastos) chartGastos.update();
    if (chartCategorias) chartCategorias.update();
}

/* Helper de preenchimento dos selects */
function preencherSelectsFormularios() {
    const selVeiculos = `<option value="">Selecione o Caminhão</option>` + db.veiculos.map(v => `<option value="${v.id}">${v.placa} - ${v.prefixo} (${v.modelo})</option>`).join('');
    document.getElementById('oVeiculoId').innerHTML = selVeiculos;
    document.getElementById('osVeiculoId').innerHTML = selVeiculos;

    const selMec = `<option value="">Selecione o Mecânico</option>` + db.mecanicos.map(m => `<option value="${m.id}">${m.nome} (${m.empresa_oficina})</option>`).join('');
    document.getElementById('osMecanicoId').innerHTML = selMec;

    const selOc = `<option value="">Selecione a Ocorrência</option>` + db.ocorrencias.filter(o => o.status !== 'concluida').map(o => `<option value="${o.id}">#${o.numero_ocorrencia} - ${o.categoria.toUpperCase()}</option>`).join('');
    document.getElementById('osOcorrenciaId').innerHTML = selOc;
}
